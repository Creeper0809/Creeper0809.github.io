## 문제 태그

- 그래프 이론
- 누적 합
- 이분 탐색

## 아이디어

1. 이 문제는 그래프 문제처럼 보이지만 이분 탐색과 누적합의 성격이 더 강함
2. 이분 탐색을 적용하려면, 단조성을 보여야 한다. 이 문제는 두 가지 측면에서 단조성을 보인다
    1. 설치 가능한 크기: 크기가 커질수록 "설치 가능"에서 "설치 불가능"으로 바뀜
    2. 경로 차단 가능한 크기: 크기가 커질수록 "차단 불가능"에서 "차단 가능"으로 바뀜
3. 각 지점에서 이 두 가지를 확인한다
    1. 예시 1: 최대 설치 가능 크기가 4이고 크기 3으로 경로를 막을 수 있다면, 크기 3으로 차단 가능
    2. 예시 2: 최대 설치 가능 크기가 1이고 크기 5로 경로를 막을 수 있다면, 차단 불가능
4. 장애물 설치와 경로 차단 여부를 매번 BFS로 확인하면 시간이 오래 걸린다. 상수 시간에 확인하기 위해 누적합을 사용한다
    1. 구간 (i,j)~(i+크기,j+크기)의 점 개수가 정사각형의 넓이와 같다면, 해당 크기로 설치가 가능하다
5. 경로 차단 여부도 BFS 대신 다음 규칙으로 확인할 수 있다
    1. 장애물이 왼쪽·아래 벽과 오른쪽·위 벽에 연결되어 있는가? 연결되어 있으면 경로가 차단된 것이다
    2. 이것도 누적합으로 구간 (i,j)~(i+크기,j+크기)의 각 왼쪽·아래 벽과 오른쪽·위 벽 #개수가 1 이상이면 연결된 것이다

주의.

1. (1,1)과 (M,N)은 설치 불가능 구역으로 처리해야 함
2. 구간합 계산 시 범위를 clamp(x,low,high)로 제한해야 함
3. 전처리로 BFS를 실행해 진입 불가능한 지점은 벽으로 처리해야 경로 차단 확인이 수월함
4. 블록 간 연결 확인은 대각선을 포함한 8방향이지만, 진입 불가능한 구간 확인용 BFS는 4방향임

다이아 난이도에 걸맞은 문제였음.

## 정답

```python
import sys
from collections import deque

input = sys.stdin.readline

DIR4 = ((0, 1), (1, 0), (0, -1), (-1, 0))
DIR8 = DIR4 + ((1, 1), (1, -1), (-1, 1), (-1, -1))

def bfs(grid, sources, target, directions):
    n, m = len(grid), len(grid[0])
    visited = [[0] * m for _ in range(n)]
    q = deque(sources)

    for r, c in sources:
        visited[r][c] = 1

    while q:
        r, c = q.popleft()

        for dr, dc in directions:
            nr, nc = r + dr, c + dc

            if not (0 <= nr < n and 0 <= nc < m):
                continue
            if visited[nr][nc] or grid[nr][nc] != target:
                continue

            visited[nr][nc] = 1
            q.append((nr, nc))

    return visited

def build_psum(grid):
    n, m = len(grid), len(grid[0])
    ps = [[0] * (m + 1) for _ in range(n + 1)]

    for i in range(n):
        row_sum = 0
        for j in range(m):
            row_sum += grid[i][j]
            ps[i + 1][j + 1] = ps[i][j + 1] + row_sum

    return ps

def query(ps, r1, c1, r2, c2):
    n, m = len(ps) - 1, len(ps[0]) - 1

    r1 = max(0, min(r1, n - 1))
    c1 = max(0, min(c1, m - 1))
    r2 = max(0, min(r2, n - 1))
    c2 = max(0, min(c2, m - 1))

    if r1 > r2 or c1 > c2:
        return 0

    return (
        ps[r2 + 1][c2 + 1]
        - ps[r1][c2 + 1]
        - ps[r2 + 1][c1]
        + ps[r1][c1]
    )

def get_sources(reachable):
    n, m = len(reachable), len(reachable[0])
    upper, lower = [], []

    for j in range(m - 1):
        if not reachable[0][j]:
            upper.append((0, j))
        if not reachable[n - 1][j]:
            lower.append((n - 1, j))

    for i in range(n - 1):
        if not reachable[i][m - 1]:
            upper.append((i, m - 1))
        if not reachable[i][0]:
            lower.append((i, 0))

    return upper, lower

def get_min_size(r, c, n, m, upper, lower):
    limit = min(n - r, m - c)
    lo, hi = 1, limit + 1

    while lo < hi:
        mid = (lo + hi) // 2

        r1, c1 = r - 1, c - 1
        r2, c2 = r + mid, c + mid

        up = (
            c + mid == m
            or query(upper, r1, c1, r2, c2) > 0
        )
        down = (
            r + mid == n
            or query(lower, r1, c1, r2, c2) > 0
        )

        if (up and down) or (up and c == 0) or (down and r == 0):
            hi = mid
        else:
            lo = mid + 1

    return lo

def solve(grid):
    n, m = len(grid), len(grid[0])

    reachable = bfs(grid, [(0, 0)], '.', DIR4)
    upper_src, lower_src = get_sources(reachable)

    upper = bfs(reachable, upper_src, 0, DIR8)
    lower = bfs(reachable, lower_src, 0, DIR8)

    reachable[n - 1][m - 1] = 0

    upper_ps = build_psum(upper)
    lower_ps = build_psum(lower)
    install_ps = build_psum(reachable)

    ans = float('inf')
    ans_r = ans_c = -1

    for r in range(n):
        for c in range(m):
            if (r, c) in ((0, 0), (n - 1, m - 1)):
                continue
            if not reachable[r][c]:
                continue

            size = get_min_size(
                r, c, n, m, upper_ps, lower_ps
            )

            if size > min(n - r, m - c) or size >= ans:
                continue

            total = query(
                install_ps,
                r, c,
                r + size - 1, c + size - 1
            )

            if total == size * size:
                ans = size
                ans_r, ans_c = r, c

    if ans == float('inf'):
        return "Impossible"

    return f"{ans} {ans_c + 1} {ans_r + 1}"

m, n = map(int, input().split())
grid = [input().strip() for _ in range(n)]

print(solve(grid))
```
