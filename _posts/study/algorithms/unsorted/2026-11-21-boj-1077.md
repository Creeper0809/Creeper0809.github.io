## 문제 태그

- 기하학
- 볼록 껍질
- 선분 교차 판정
- 다각형의 넓이
- 볼록 다각형 내부의 점 판정

## 아이디어

- 두 번째 볼록 다각형의 각 변을 기준으로 안쪽 반평면만 남긴다. 볼록 다각형은 이 반평면들의 교집합이므로 첫 번째 다각형을 변마다 잘라 가면 둘이 겹치는 부분만 남는다.
- 잘리는 다각형의 점을 순서대로 보면서, 두 점이 모두 안쪽이면 다음 점을 넣는다. 경계를 건너는 경우에는 교점을 넣고, 안으로 들어오는 경우에는 다음 점도 함께 넣는다. 안쪽인지는 외적의 부호로 판정한다.
- 남은 꼭짓점을 순서대로 신발끈 공식에 넣는다. 점이 3개 미만이면 넓이는 0이고, 한 다각형이 다른 다각형 안에 전부 들어 있으면 안쪽 다각형의 넓이가 그대로 나온다.

예를 들어 가로 범위가 각각 0~2와 1~3이고 세로 범위가 각각 0~2와 1~3인 정사각형을 겹치면 두 축 모두 1~2만 남는다. 교집합의 넓이는 1이다.

## 정답

```python
import sys
from typing import Tuple, List
input = sys.stdin.readline

Point = Tuple[float, float]
EPS = 1e-9

def cross(x1: float, y1: float, x2: float, y2: float) -> float:
    return x1 * y2 - y1 * x2

def orientation(a: Point, b: Point, c: Point) -> int:
    v = cross(b[0] - a[0], b[1] - a[1], c[0] - a[0], c[1] - a[1])
    if abs(v) < EPS:
        return 0
    return 1 if v < 0 else 2

def on_segment(a: Point, b: Point, c: Point) -> bool:
    return (min(a[0], c[0]) - EPS <= b[0] <= max(a[0], c[0]) + EPS and
            min(a[1], c[1]) - EPS <= b[1] <= max(a[1], c[1]) + EPS)

def segments_intersect(p1: Point, p2: Point, p3: Point, p4: Point) -> bool:
    o1 = orientation(p1, p2, p3)
    o2 = orientation(p1, p2, p4)
    o3 = orientation(p3, p4, p1)
    o4 = orientation(p3, p4, p2)
    if o1 != o2 and o3 != o4:
        return True
    if o1 == 0 and on_segment(p1, p3, p2):
        return True
    if o2 == 0 and on_segment(p1, p4, p2):
        return True
    if o3 == 0 and on_segment(p3, p1, p4):
        return True
    if o4 == 0 and on_segment(p3, p2, p4):
        return True
    return False

def intersection_point(p1: Point, p2: Point, p3: Point, p4: Point) -> Point:
    denom = (p1[0] - p2[0]) * (p3[1] - p4[1]) - (p1[1] - p2[1]) * (p3[0] - p4[0])
    if abs(denom) < EPS:
        return None
    t = ((p1[0] - p3[0]) * (p3[1] - p4[1]) - (p1[1] - p3[1]) * (p3[0] - p4[0])) / denom
    return (p1[0] + t * (p2[0] - p1[0]), p1[1] + t * (p2[1] - p1[1]))

def is_inside_clip(p: Point, clip_start: Point, clip_end: Point) -> bool:
    return orientation(clip_start, clip_end, p) != 1

def sutherland_hodgman(subject: List[Point], clip: List[Point]) -> List[Point]:
    output = subject[:]
    for i in range(len(clip)):
        clip_start = clip[i]
        clip_end = clip[(i + 1) % len(clip)]
        input_list = output[:]
        output = []
        for j in range(len(input_list)):
            curr = input_list[j]
            prev = input_list[(j - 1) % len(input_list)]
            if is_inside_clip(curr, clip_start, clip_end):
                if not is_inside_clip(prev, clip_start, clip_end):
                    inter = intersection_point(prev, curr, clip_start, clip_end)
                    if inter:
                        output.append(inter)
                output.append(curr)
            elif is_inside_clip(prev, clip_start, clip_end):
                inter = intersection_point(prev, curr, clip_start, clip_end)
                if inter:
                    output.append(inter)
    return output

def shoelace_formula(points, n):
    area = 0.0
    for i in range(n):
        j = (i + 1) % n
        area += points[i][0] * points[j][1]
        area -= points[i][1] * points[j][0]
    area = abs(area) / 2.0
    return area

def main():
    N, M = map(int, input().split())
    poly1 = []
    for _ in range(N):
        x, y = map(float, input().split())
        poly1.append((x, y))
    poly2 = []
    for _ in range(M):
        x, y = map(float, input().split())
        poly2.append((x, y))
    inter_poly = sutherland_hodgman(poly1, poly2)
    n = len(inter_poly)
    if n < 3:
        print("0.0000000000")
    else:
        area = shoelace_formula(inter_poly, n)
        print(f"{area:.10f}")

if __name__ == "__main__":
    main()
```
