## 문제 태그

- 그래프 이론
- 최대 유량
- 최소 비용 최대 유량

## 아이디어

- 승객 대신 좌석 한 자리를 유량 한 단위로 생각한다. 연속된 역을 잇는 간선의 비용은 0이고, 용량은 열차의 정원 P다. 이 간선은 해당 구간을 빈 좌석으로 지나가는 경우다.
- i역에서 j역까지 가는 승객은 i에서 j로 바로 건너뛰는 간선으로 만든다. 용량은 탈 수 있는 승객 수, 비용은 운임의 음수다. 한 좌석의 경로에서 이 간선을 선택하면 j역까지 그 좌석을 이미 사용 중이라 중간에서 다른 승객을 태울 수 없다.
- 정원만큼 유량을 흘리면서 비용을 최소로 만든다. SPFA로 잔여 그래프의 최단 경로를 찾고, 그 경로의 최소 잔여 용량만큼 보낸다. 마지막 비용에 마이너스를 붙이면 최대 수익이 된다.

좌석이 하나인 열차에서 1→3 승객의 요금이 8이고, 1→2와 2→3 승객 요금이 각각 5와 6이라면 두 구간으로 나눠 태워 11을 받는 쪽이 낫다.

## 정답

```python
import sys
from collections import deque


class Edge:
    def __init__(self, to, cap, cost):
        self.to = to
        self.cap = cap
        self.cost = cost
        self.flow = 0
        self.rev = None


def add_edge(graph, fr, to, cap, cost):
    forward = Edge(to, cap, cost)
    backward = Edge(fr, 0, -cost)
    forward.rev = backward
    backward.rev = forward
    graph[fr].append(forward)
    graph[to].append(backward)


def spfa(graph, s, t, dist, parent):
    n = len(graph)
    in_queue = [False] * n
    dist[:] = [float('inf')] * n
    dist[s] = 0
    queue = deque([s])
    in_queue[s] = True
    while queue:
        u = queue.popleft()
        in_queue[u] = False
        for e in graph[u]:
            if e.cap - e.flow > 0 and dist[e.to] > dist[u] + e.cost:
                dist[e.to] = dist[u] + e.cost
                parent[e.to] = (u, e)
                if not in_queue[e.to]:
                    in_queue[e.to] = True
                    queue.append(e.to)
    return dist[t] != float('inf')


def min_cost_max_flow(graph, s, t):
    flow = 0
    cost = 0
    n = len(graph)
    dist = [0] * n
    parent = [None] * n

    while spfa(graph, s, t, dist, parent):
        path_flow = float('inf')
        v = t
        while v != s:
            u, e = parent[v]
            path_flow = min(path_flow, e.cap - e.flow)
            v = u

        v = t
        while v != s:
            u, e = parent[v]
            e.flow += path_flow
            e.rev.flow -= path_flow
            cost += path_flow * e.cost
            v = u

        flow += path_flow

    return flow, cost


def solve():
    N, P = map(int, input().split())

    passengers_A = []
    for _ in range(N - 1):
        passengers_A.append(list(map(int, sys.stdin.readline().strip().split())))

    fares_C = []
    for _ in range(N - 1):
        fares_C.append(list(map(int, sys.stdin.readline().strip().split())))

    num_nodes = N + 1
    source = 0
    sink = N
    graph = [[] for _ in range(num_nodes)]

    for i in range(N):
        add_edge(graph, i, i + 1, P, 0)

    for i_idx in range(N - 1):
        start_city = i_idx + 1
        for j_idx in range(len(passengers_A[i_idx])):
            j = j_idx + 1
            end_city = start_city + j

            if end_city > N: continue

            count = passengers_A[i_idx][j_idx]
            fare = fares_C[i_idx][j_idx]

            if count > 0:
                fr = start_city - 1
                to = end_city - 1
                add_edge(graph, fr, to, count, -fare)

    _, min_cost = min_cost_max_flow(graph, source, sink)

    max_profit = -min_cost
    print(max_profit)


solve()
```
