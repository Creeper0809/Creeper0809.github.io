## 문제 태그

- DP
- 자료구조
- 트리
- CHT

## 아이디어

- dp 배열 정의
    - dp[u] = 도시 u에서 수도 까지 메시지를 전달하는 데 걸리는 최소 시간
- 모든 도시 i에 대해 dp[i]를 구하는것이 목적
- dp[u]는 다음과 같이 모든 가능한 선택지 중 최소값으로 결정 됨
    - 도시 u의 전령이 조상 도시 v까지 메시지를 직접 전달하고 도시 v에서부터는 최적의 방법(dp[v])로 수도까지 전달
- 총 소요 시간 계산
    - 도시 u의 전령이 준비하는데엔 $$S_u$$의 시간이 걸림
    - 도시 u의 전령이 u→v까지 이동하는데에 거리는 dist(1,u) - dist(1,v), u 전령의 속도는 $$V_u$$고로 이동시간은 $$V_u \times (dist(1, u) - dist(1, v))$$
    - 도시 v에 도착한 후 메시지는 v에서 수도까지 dp로 이미 계산이 됐다고 가정, dp[v]
- 점화식으로 표현하면
    
    $$
    dp[u] = \underset{v\in ancestors(u)}{\text{min}}(S_u + V_u \cdot (dist[u] - dist[v]) + dp[v])
    $$
    
    하지만 이런 점화식으로는 매번 모든 조상을 순회해야함으로 비효율적임
    
    우선 min 내부 항들을 v에 대하여 나타내어준다
    
    $$
    dp[u] = S_u + V_u \cdot dist[u] + \underset{v\in ancestors(u)}{\text{min}}( dp[v] - V_u \cdot dist[v])
    $$
    
    각 조상 v에 대해 $$m_v = dist[v]$$와 $$b_v = dp[v]$$를 알고 있다면 위 식을 다음과 같이 변형할 수 있음
    
    $$
    dp[u] = S_u + V_u \cdot dist[u] + \underset{v\in ancestors(u)}{\text{min}}(-m_v \cdot V_u + b_v)
    $$
    
    이는 직선의 방정식이며 기울기가 감소함에 CHT를 적용하여 최적화를 할 수 있음을 알 수 있다.
    

## 정답

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int MAXN = 100010;

struct Line {
    ll m, b;
    Line() : m(0), b(0) {}
    Line(ll m_, ll b_) : m(m_), b(b_) {}
};

vector<Line> lines(MAXN);

int ptr = 1;

ll eval(const Line& l, ll x) {
    return (ll)((__int128)l.m * x + l.b);
}

bool q_check(const Line& A, const Line& B, ll x) {
    __int128 left = (__int128)(B.b - A.b);
    __int128 right = (__int128)x * (A.m - B.m);
    return left < right;
}

ll query(ll x) {
    int lo = 0, hi = ptr - 1;
    while (lo < hi) {
        int mid = (lo + hi + 1) / 2;
        Line A = lines[mid - 1];
        Line B = lines[mid];
        if (q_check(A, B, x)) {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    return eval(lines[lo], x);
}

bool add_check(const Line& A, const Line& B, const Line& C) {
    __int128 p = (__int128)(A.b - C.b);
    __int128 q = (__int128)(B.m - A.m);
    __int128 r = (__int128)(A.b - B.b);
    __int128 s = (__int128)(C.m - A.m);
    return p * q < r * s;
}

int add_line(Line new_l) {
    int lo = 1, hi = ptr;
    while (lo < hi) {
        int mid = (lo + hi) / 2;
        Line A = lines[mid - 1];
        Line B = lines[mid];
        if (add_check(A, B, new_l)) {
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }
    return lo;
}

vector<pair<int, ll>> adj[MAXN];
ll S[MAXN], V[MAXN], f[MAXN], dist[MAXN];

struct RollbackInfo {
    int rb_idx;
    Line rb_line;
    int old_ptr;
};

RollbackInfo add_line_rb(ll new_m, ll new_b) {
    Line cur_line(new_m, new_b);
    int old_ptr = ptr;
    int rb_idx = add_line(cur_line);
    Line rb_line = lines[rb_idx];
    lines[rb_idx] = cur_line;
    ptr = rb_idx + 1;
    return {rb_idx, rb_line, old_ptr};
}

void undo_add(const RollbackInfo& info) {
    lines[info.rb_idx] = info.rb_line;
    ptr = info.old_ptr;
}

void dfs(int u, int par, ll cur_dist) {
    dist[u] = cur_dist;
    RollbackInfo info = {0, Line(), 0};
    if (u != 1) {
        ll min_val = query(V[u]);
        f[u] = S[u] + dist[u] * V[u] + min_val;
        info = add_line_rb(-dist[u], f[u]);
    }
    for (auto& p : adj[u]) {
        int v = p.first;
        ll d = p.second;
        if (v != par) {
            dfs(v, u, cur_dist + d);
        }
    }
    if (u != 1) {
        undo_add(info);
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int N;
    cin >> N;
    for (int i = 0; i < N - 1; i++) {
        int u, v, d;
        cin >> u >> v >> d;
        adj[u].emplace_back(v, d);
        adj[v].emplace_back(u, d);
    }
    for (int i = 2; i <= N; i++) {
        cin >> S[i] >> V[i];
    }
    lines[0] = Line(0, 0);
    dfs(1, -1, 0);
    for (int i = 2; i <= N; i++) {
        if (i > 2) cout << " ";
        cout << f[i];
    }
    cout << "\n";
    return 0;
}
```
